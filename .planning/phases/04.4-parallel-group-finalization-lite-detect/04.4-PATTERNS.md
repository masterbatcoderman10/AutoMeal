    self,
    *,
    image_data_url: str,
    user_id: str,
    model: str | None = None,
    confidence_threshold: float | None = None,
) -> DetectFoodResult:

        confidence_threshold

        else settings.FOOD_DETECT_CONFIDENCE_THRESHOLD

```python
async def _resolve_match_parallelism(self, *, max_concurrency: int) -> None:
    sem = asyncio.Semaphore(max_concurrency)

    async def _match_segment_with_isolated_session(segment):
        async with sem:
            return await self._match_one_segment(segment)

    return await asyncio.gather(*[
        _match_segment_with_isolated_session(s) for s in segments
    ])
```
